const SHEET_NAME = "Scans";

function doGet() {
  const sheet = getSheet();
  const rows = sheet.getDataRange().getValues();
  let html = "<h1>AmeriLanka Saved Scans</h1>";

  if (rows.length <= 1) {
    html += "<p>No saved scans yet.</p>";
  } else {
    for (let i = rows.length - 1; i >= 1; i--) {
      const row = rows[i];
      const copyBox = row[7] || buildCopyBox(row);
      html += "<div style='border:1px solid #ddd;border-radius:8px;padding:12px;margin:12px 0;background:#fff;'>";
      html += "<div style='font-size:12px;color:#666;margin-bottom:8px;'>" + safe(row[0]) + "</div>";
      html += "<pre style='font-family:Arial,sans-serif;font-weight:bold;white-space:pre-wrap;line-height:1.5;margin:0;'>" + safe(copyBox) + "</pre>";
      html += "</div>";
    }
  }

  html = "<html><body style='font-family:Arial,sans-serif;background:#f4f7fb;padding:20px;max-width:760px;margin:auto;'>" + html + "</body></html>";
  return HtmlService.createHtmlOutput(html);
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents || "{}");
    if (data.action === "docfill") {
      return sendJson(createFilledDoc(data));
    }

    const copyBox = data.copyBox || makeCopyBox(data);
    getSheet().appendRow([
      data.savedAt || new Date().toISOString(),
      data.surname || "",
      data.givenNames || "",
      data.birthDate || "",
      data.gender || "",
      data.nationality || "",
      data.expirationDate || "",
      copyBox
    ]);

    return sendJson({ ok: true });
  } catch (error) {
    return sendJson({ ok: false, error: error.message || String(error) });
  }
}

function createFilledDoc(data) {
  const props = PropertiesService.getScriptProperties();
  const templateId = props.getProperty("DOCFILL_TEMPLATE_ID");
  const folderId = props.getProperty("DOCFILL_FOLDER_ID");

  if (!templateId) {
    throw new Error("DOCFILL_TEMPLATE_ID is not set in Apps Script properties.");
  }

  const values = docFillValues(data);
  const fileName = "DocFill-" + values.A3.replace(/[^A-Z0-9]+/g, "-").replace(/^-|-$/g, "") + "-" + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyyMMdd-HHmmss");
  const templateFile = DriveApp.getFileById(templateId);
  const folder = folderId ? DriveApp.getFolderById(folderId) : DriveApp.getRootFolder();
  const copy = templateFile.makeCopy(fileName, folder);
  const doc = DocumentApp.openById(copy.getId());
  const body = doc.getBody();

  replacePlaceholder(body, "A3", values.A3);
  replacePlaceholder(body, "B3", values.B3);
  replacePlaceholder(body, "C3", values.C3);
  replacePlaceholder(body, "A10", values.A10);
  replacePlaceholder(body, "COPYBOX", values.copyBox);
  doc.saveAndClose();

  const exportUrl = "https://www.googleapis.com/drive/v3/files/" + copy.getId() + "/export?mimeType=application/vnd.openxmlformats-officedocument.wordprocessingml.document";
  const response = UrlFetchApp.fetch(exportUrl, {
    headers: { Authorization: "Bearer " + ScriptApp.getOAuthToken() }
  });
  const blob = response.getBlob().setName(fileName + ".docx");

  return {
    ok: true,
    docUrl: doc.getUrl(),
    fileName: blob.getName(),
    mimeType: blob.getContentType(),
    fileBase64: Utilities.base64Encode(blob.getBytes())
  };
}

function docFillValues(data) {
  const last = cleanName(data.surname || "");
  const firstMiddle = cleanName(data.givenNames || "");
  const dob = travelDate(data.birthDate || "");
  const gender = genderText(data.gender || "");
  const expire = expiryYear(data.expirationDate || "");
  const country = countryCode(data.nationality || "");

  return {
    A3: [last, firstMiddle].filter(Boolean).join("/"),
    B3: [dob, gender].filter(Boolean).join("/"),
    C3: [expire, country].filter(Boolean).join("/"),
    A10: ["3DOCSA", "DB", dob, gender, last, firstMiddle].filter(Boolean).join("/"),
    copyBox: makeCopyBox(data)
  };
}

function replacePlaceholder(body, key, value) {
  body.replaceText("\\{\\{" + key + "\\}\\}", value || "");
}

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Saved At", "Surname", "Other Name / Given Name", "Birth Date", "Gender", "Nationality", "Expiration Date", "Copy Box"]);
    sheet.setFrozenRows(1);
    sheet.getRange("H:H").setWrap(true);
  }
  return sheet;
}

function buildCopyBox(row) {
  return [
    cleanName(row[1] || ""),
    cleanName(row[2] || ""),
    travelDate(row[3] || ""),
    genderText(row[4] || ""),
    expiryYear(row[6] || ""),
    countryCode(row[5] || "")
  ].filter(Boolean).join("/");
}

function makeCopyBox(data) {
  return [
    cleanName(data.surname || ""),
    cleanName(data.givenNames || ""),
    travelDate(data.birthDate || ""),
    genderText(data.gender || ""),
    expiryYear(data.expirationDate || ""),
    countryCode(data.nationality || "")
  ].filter(Boolean).join("/");
}

function genderText(value) {
  const gender = String(value || "").toUpperCase();
  if (gender === "M" || gender === "MALE") return "M";
  if (gender === "F" || gender === "FEMALE") return "F";
  return value || "";
}

function cleanName(value) {
  return String(value || "").toUpperCase().replace(/[^A-Z .'-]/g, " ").replace(/\s+/g, " ").trim();
}

function travelDate(value) {
  const parts = dateParts(value);
  if (!parts) return "";
  return String(parts.day).padStart(2, "0") + monthAbbrev(parts.month) + String(parts.year).slice(-2);
}

function expiryYear(value) {
  const parts = dateParts(value);
  return parts ? String(parts.year).slice(-2) : "";
}

function dateParts(value) {
  const text = String(value || "").trim();
  let match = text.match(/^(\d{1,2})\s+([A-Za-z]+)\s+(\d{2}|\d{4})$/);
  if (match) return { day: Number(match[1]), month: monthNumber(match[2]), year: match[3] };
  match = text.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (match) return { day: Number(match[3]), month: Number(match[2]), year: match[1] };
  match = text.match(/^(\d{1,2})[\/\-. ](\d{1,2})[\/\-. ](\d{2}|\d{4})$/);
  if (match) return { day: Number(match[1]), month: Number(match[2]), year: match[3] };
  return null;
}

function monthAbbrev(value) {
  const names = ["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"];
  return names[Number(value) - 1] || "";
}

function monthNumber(value) {
  const names = ["january", "february", "march", "april", "may", "june", "july", "august", "september", "october", "november", "december"];
  const month = names.indexOf(String(value || "").toLowerCase());
  return month >= 0 ? month + 1 : 0;
}

function countryCode(value) {
  const compact = String(value || "").toUpperCase().replace(/[^A-Z]/g, "");
  const map = {
    USA: "US",
    US: "US",
    UNITEDSTATES: "US",
    UNITEDSTATESOFAMERICA: "US",
    AMERICAN: "US",
    LKA: "LK",
    LK: "LK",
    SRILANKA: "LK",
    SRILANKAN: "LK"
  };
  if (map[compact]) return map[compact];
  return compact.slice(0, 2);
}

function sendJson(payload) {
  return ContentService.createTextOutput(JSON.stringify(payload)).setMimeType(ContentService.MimeType.JSON);
}

function safe(value) {
  return String(value || "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}
